## 四、核对方法

```bash
cd "GitHub 发布版本"

# 1) 文件数与 config
find EFI -type f | wc -l                       # 205
shasum -a 256 EFI/OC/config.plist              # bdf0cbea…
plutil -lint EFI/OC/config.plist               # OK

# 2) 逐文件核对 EFI（清单里是相对 EFI/ 的路径）
python3 - <<'PY'
import hashlib, pathlib
root = pathlib.Path("EFI")
exp = {}
for l in pathlib.Path("文档/EFI文件清单-SHA256.txt").read_text().splitlines():
    if l.startswith("#") or not l.strip(): continue
    h, rel = l.split("  ", 1)
    exp[rel] = h
got = {str(p.relative_to(root)): hashlib.sha256(p.read_bytes()).hexdigest()
       for p in root.rglob("*") if p.is_file()}
print("缺失:", sorted(set(exp)-set(got)))
print("多余:", sorted(set(got)-set(exp)))
print("不符:", sorted(k for k in exp.keys() & got.keys() if exp[k] != got[k]))
PY

# 3) 全包扫描真实三码
#    把下面四个占位换成你自己那份自用 config 里的真值，再执行；应无任何输出
python3 - <<'PY'
import pathlib
NEEDLES = [b"<自用序列号>", b"<自用MLB>", b"<自用UUID>", b"<自用ROM十六进制>"]
hit = []
for p in pathlib.Path(".").rglob("*"):
    if not p.is_file() or p.suffix == ".zip": continue
    data = p.read_bytes()
    if any(n in data for n in NEEDLES): hit.append(str(p))
print("命中:", hit if hit else "零命中")
PY
```
